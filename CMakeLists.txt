cmake_minimum_required(VERSION 3.24)
project(Vibmory VERSION 2.0.0 LANGUAGES CXX)

include(CTest)
if(WIN32)
    set(CMAKE_INSTALL_BINDIR "." CACHE PATH "Portable executable and runtime directory")
endif()
include(GNUInstallDirs)
option(VIBMORY_BUILD_GUI "Build the Qt 6 desktop editor" ON)
option(VIBMORY_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(VIBMORY_SANITIZERS "Enable address/undefined sanitizers (Clang/GCC)" OFF)
option(VIBMORY_BUILD_BENCHMARKS "Build informational renderer benchmarks" OFF)

add_library(vibmory_options INTERFACE)
target_compile_features(vibmory_options INTERFACE cxx_std_20)
if(MSVC)
    target_compile_options(vibmory_options INTERFACE /W4 /w15038 /permissive- /utf-8 /Zc:__cplusplus)
    if(VIBMORY_WARNINGS_AS_ERRORS)
        target_compile_options(vibmory_options INTERFACE /WX)
    endif()
else()
    target_compile_options(vibmory_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
    if(VIBMORY_WARNINGS_AS_ERRORS)
        target_compile_options(vibmory_options INTERFACE -Werror)
    endif()
    if(VIBMORY_SANITIZERS)
        target_compile_options(vibmory_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(vibmory_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

function(vibmory_library name)
    add_library(${name} STATIC ${ARGN})
    target_include_directories(${name} PUBLIC "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${name} PUBLIC cxx_std_20)
    target_link_libraries(${name} PRIVATE vibmory_options)
    set_target_properties(${name} PROPERTIES CXX_EXTENSIONS OFF AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
endfunction()

vibmory_library(vibmory_domain src/domain/pattern.cpp)
vibmory_library(vibmory_application src/application/editor_session.cpp src/application/playback_service.cpp src/application/document_workflow.cpp)
target_link_libraries(vibmory_application PUBLIC vibmory_domain)
vibmory_library(vibmory_audio src/audio/pcm_renderer.cpp)
target_link_libraries(vibmory_audio PUBLIC vibmory_domain)
vibmory_library(vibmory_persistence src/persistence/project_codec.cpp)
target_link_libraries(vibmory_persistence PUBLIC vibmory_domain)

if(VIBMORY_BUILD_BENCHMARKS)
    add_executable(vibmory_render_benchmark tools/render_benchmark.cpp)
    target_link_libraries(vibmory_render_benchmark PRIVATE vibmory_audio vibmory_options)
endif()

if(VIBMORY_BUILD_GUI)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Multimedia Concurrent)
    qt_standard_project_setup()
    vibmory_library(vibmory_platform src/platform/project_store.cpp src/platform/qt_audio_backend.cpp src/platform/wave_export_job.cpp)
    set_target_properties(vibmory_platform PROPERTIES AUTOMOC ON)
    target_link_libraries(vibmory_platform PUBLIC vibmory_application vibmory_audio vibmory_persistence Qt6::Core Qt6::Multimedia PRIVATE Qt6::Concurrent)
    vibmory_library(vibmory_ui src/ui/grid_widget.cpp src/ui/cell_inspector.cpp src/ui/file_dialogs.cpp src/ui/frame_list_model.cpp src/ui/editor_window.cpp)
    set_target_properties(vibmory_ui PROPERTIES AUTOMOC ON)
    target_link_libraries(vibmory_ui PUBLIC vibmory_application Qt6::Widgets PRIVATE vibmory_platform)
    qt_add_executable(Vibmory WIN32 src/app/main.cpp)
    set_target_properties(Vibmory PROPERTIES AUTOMOC OFF AUTOUIC OFF CXX_EXTENSIONS OFF)
    if(WIN32)
        enable_language(RC)
        configure_file(src/app/windows.rc.in "${PROJECT_BINARY_DIR}/windows.rc" @ONLY)
        target_sources(Vibmory PRIVATE "${PROJECT_BINARY_DIR}/windows.rc")
    endif()
    target_link_libraries(Vibmory PRIVATE vibmory_ui vibmory_platform vibmory_options)
    target_compile_definitions(Vibmory PRIVATE VIBMORY_VERSION="${PROJECT_VERSION}")
    install(TARGETS Vibmory RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" BUNDLE DESTINATION .)
    install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
    qt_generate_deploy_app_script(TARGET Vibmory OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()

find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
    configure_file(Doxyfile "${PROJECT_BINARY_DIR}/Doxyfile" @ONLY)
    add_custom_target(docs COMMAND Doxygen::doxygen "${PROJECT_BINARY_DIR}/Doxyfile"
        WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" COMMENT "Generate architecture and API documentation")
endif()

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
